export const stats = memo(function stats(img) {
  const L = img.length, s = new Float32Array(4000);
  for (let k = 0; k < 4000; k++) s[k] = img[(k * 25733) % L];
  s.sort();
  const med = s[2000], d = new Float32Array(4000);
  for (let k = 0; k < 4000; k++) d[k] = Math.abs(s[k] - med);
  d.sort();
  return { med, sig: 1.4826 * d[2000] };
});

// The "single-pixel spike" test compares a peak with the mean of its 8 neighbours. What counts as too
// sharp depends on how well the PSF is sampled: the simulation's σ = 1.15 px Gaussian gives ≈ 1.7 for a
// centred star, but real SPHEREx stars (FWHM ≈ 6.6″ ≈ 1.07 px at 6.15″/px) give 3–8. So the limit is
// scaled by the PSF: spikeLimit(σ) = 3.2 · S(σ) / S(1.15), where S(σ) is the spike ratio of a centred,
// point-sampled Gaussian.
export const SIM_PSF_SIGMA = 1.15;
export const SPIKE_LIMIT_SIM = 3.2;

/** Spike ratio (peak / mean of the 8 neighbours) of a centred, point-sampled Gaussian of width σ px. */
export function spikeOf(sigma) {
  const u = Math.exp(-1 / (2 * sigma * sigma));
  return 2 / (u + u * u);
}

/** Spike limit for a PSF of width σ px. */
export function spikeLimit(sigma) {
  return sigma === SIM_PSF_SIGMA ? SPIKE_LIMIT_SIM : SPIKE_LIMIT_SIM * spikeOf(sigma) / spikeOf(SIM_PSF_SIGMA);
}
